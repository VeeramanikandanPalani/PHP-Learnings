#Migration
php artisan migrate - run migration
php artisan make:migration create_students_table // This is the proper way for creating migration
php artisan make:migration addDateOfBirthToStudents --table=students //For adding missing columns

#Rollback our latest migration
php artisan migrate:rollback //Reverse latest one
php artisan migrate:rollback --step=2 //It will remove last 2 migrations
php artisan migrate:rollback --step=3 //Revert last 3 migrations

#Raw SQL vs Query Builder vs Eloquent: Which One Should You Use?
Raw SQL - High risk - use when need complex queries and good performance
Query Builder - Protect against sql injection - 
ORM - Highly secure - use when working with models

#Insert using Query Builder

#For single record insertion
DB::table('student')->insert([
  'name'=>'veera',
  'age'=>14
]);

#For bulk record insertion
DB::table('student')->insert([
  [
    'name'=>'veera',
    'age'=>14
  ],
  [
    'name'=>'mithra',
    'age'=>1
  ],
]);

#Fetch data using Query Builder
$Data = DB::table('students')->get();
$Data = DB::table('students')->first();
$Data = DB::table('students')
        ->select('name','age')
        ->where('id','>',1)
        ->get();

#Update using query builder
DB::table('students')->where('id',203)->update('name','updated name');

#Delete using query builder

#WhereAny

#WhereAll
