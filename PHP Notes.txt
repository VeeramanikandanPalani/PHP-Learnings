#Route prefix

  Route::get('list',function(){
    return view('welcome);
  })->name('user-list');
});

#Route parameter
Route::get('student/{id} ',function(){
    return view('welcome);
})->name('user-list');

  return "Route not available'; 
});

#Access to view
Route::view('contact','contact');

#Passing data via view
Route:get('about-us',function(){
  $name ='Veera'
  return view('contact')->with('name'$name); //Method 1

  return view('contact',['name'=>$name])// Method 3
});

#Controllers
- Invokable controller
- Resource controller

php artisan make:controller TestController --invokable //Only one method will be there
Route::get('contact',TestController::class);//No mthod name needed

php artisan make:controller ResourceController --resource  //Resource controller
Route::resource('contact',ResourceController::class);//No mthod name needed

#Migration
php artisan migrate - run migration
php artisan make:migration create_students_table // This is the proper way for creating migration
php artisan make:migration addDateOfBirthToStudents --table=students //For adding missing columns

#Rollback our latest migration
php artisan migrate:rollback //Reverse latest one
php artisan migrate:rollback --step=2 //It will remove last 2 migrations
php artisan migrate:rollback --step=3 //Revert last 3 migrations
